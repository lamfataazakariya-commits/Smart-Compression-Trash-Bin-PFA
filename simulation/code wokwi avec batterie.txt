#include <LiquidCrystal_I2C.h>

// --- Configuration des Broches ---
const int TRIG_PIN = 5;
const int ECHO_PIN = 18;
const int LED_ROUGE = 26;   // Descente du vérin (Compression)
const int LED_VERTE = 27;   // Remontée du vérin

LiquidCrystal_I2C lcd(0x27, 16, 2);

// --- Icônes de Batterie (Niveau de remplissage) ---
byte p0[8] = {B01110, B10001, B10001, B10001, B10001, B10001, B10001, B11111}; // Vide
byte p1[8] = {B01110, B10001, B10001, B10001, B10001, B10001, B11111, B11111}; // 25%
byte p2[8] = {B01110, B10001, B10001, B10001, B11111, B11111, B11111, B11111}; // 50%
byte p3[8] = {B01110, B10001, B11111, B11111, B11111, B11111, B11111, B11111}; // 75%
byte p4[8] = {B01110, B11111, B11111, B11111, B11111, B11111, B11111, B11111}; // 100%

// --- Variables de Gestion ---
int etatPoubelle = 0; 
unsigned long chrono = 0; 

const unsigned long TEMPS_COMPRESSION = 4000; 
const unsigned long TEMPS_REMONTEE = 4000;    
const unsigned long TEMPS_STOP = 10000;       

void setup() {
  Serial.begin(115200);
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  pinMode(LED_ROUGE, OUTPUT);
  pinMode(LED_VERTE, OUTPUT);

  lcd.init();
  lcd.backlight();

  // Enregistrement des icônes
  lcd.createChar(0, p0);
  lcd.createChar(1, p1);
  lcd.createChar(2, p2);
  lcd.createChar(3, p3);
  lcd.createChar(4, p4);

  lcd.setCursor(0, 0);
  lcd.print("  SYSTEME PFA   ");
  lcd.setCursor(0, 1);
  lcd.print(" INITIALISATION ");
  delay(2000);
  lcd.clear();
}

void loop() {
  // --- 1. Lecture de la distance ---
  digitalWrite(TRIG_PIN, LOW); delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH); delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  
  long duration = pulseIn(ECHO_PIN, HIGH, 30000);
  float distance_cm = (duration == 0) ? 400.0 : duration * 0.034 / 2;
  
  unsigned long maintenant = millis();

  // --- 2. Affichage de l'icône batterie (Basé sur tes dimensions) ---
  lcd.setCursor(15, 0); 
  if (distance_cm > 30) lcd.write(0);      // Vide (Proche du max 35cm)
  else if (distance_cm > 25) lcd.write(1); // Début de remplissage chambre
  else if (distance_cm > 21) lcd.write(2); // Mi-chambre
  else if (distance_cm > 18) lcd.write(3); // Presque plein (Proche de 17cm)
  else lcd.write(4);                       // PLEIN (Seuil 17cm atteint)

  // --- 3. Logique de la Machine à États ---
  switch(etatPoubelle) {
    
    case 0: // ÉTAT 0 : EN ATTENTE
      digitalWrite(LED_ROUGE, LOW);
      digitalWrite(LED_VERTE, LOW);
      lcd.setCursor(0, 0); lcd.print("Niveau: "); lcd.print(distance_cm, 1); lcd.print(" cm ");
      lcd.setCursor(0, 1); lcd.print("En attente.     ");

      // Déclenchement à 5cm du haut
      if (distance_cm <= 5.0) {
        etatPoubelle = 1;      
        chrono = maintenant;   
      }
      break;

    case 1: // ÉTAT 1 : COMPRESSION
      digitalWrite(LED_ROUGE, HIGH);
      digitalWrite(LED_VERTE, LOW);
      lcd.setCursor(0, 0); lcd.print("Action: Compression");
      lcd.setCursor(0, 1); lcd.print("Verrin en bas.. ");

      if (maintenant - chrono >= TEMPS_COMPRESSION) {
        etatPoubelle = 2;      
        chrono = maintenant;   
      }
      break;

    case 2: // ÉTAT 2 : REMONTÉE
      digitalWrite(LED_ROUGE, LOW);
      digitalWrite(LED_VERTE, HIGH);
      lcd.setCursor(0, 0); lcd.print("Action: Remontee   ");
      lcd.setCursor(0, 1); lcd.print("Retour piston.. ");

      if (maintenant - chrono >= TEMPS_REMONTEE) {
        etatPoubelle = 3;      
        chrono = maintenant;   
      }
      break;

    case 3: // ÉTAT 3 : ANALYSE (10 SECONDES)
      digitalWrite(LED_VERTE, LOW);
      lcd.setCursor(0, 0); lcd.print("Analyse niveau  ");
      lcd.setCursor(0, 1); lcd.print("STOP - Attente  ");

      // Si après compression la distance reste <= 17cm, la chambre est pleine
      if (distance_cm <= 17.0) {
        etatPoubelle = 4;
      } 
      else if (maintenant - chrono >= TEMPS_STOP) {
        etatPoubelle = 0; 
      }
      break;

    case 4: // ÉTAT 4 : POUBELLE PLEINE
      digitalWrite(LED_ROUGE, LOW);
      digitalWrite(LED_VERTE, LOW);
      lcd.setCursor(0, 0); lcd.print("POUBELLE PLEINE!");
      lcd.setCursor(0, 1); lcd.print("Veuillez vider. ");

      // Réinitialisation automatique si vidée (distance > 30cm)
      if (distance_cm > 30.0) {
        etatPoubelle = 0; 
      }
      break;
  }

  delay(100); 
}
